// 継承指定と final を伴うクラスを示す。
class Base {
public:
	virtual ~Base() = default;
};

class Derived final:public Base {
public:
	explicit Derived(int value) noexcept
		:value_(value)
	{
	}

	[[nodiscard]] int operator()(int offset) const noexcept
	{
		return value_ + offset;
	}

	[[nodiscard]] int operator[](int index) const noexcept
	{
		return value_ + index;
	}

	explicit operator bool() const noexcept
	{
		return value_ != 0;
	}

	Derived& operator<<(int value) noexcept
	{
		value_ = value;
		return *this;
	}

private:
	int value_;
};

// 型制約（C++20 のコンセプト）のサンプルを示す。
template <class T>
concept Numeric = std::integral<T> || std::floating_point<T>;

// クラステンプレートと明示的特殊化を示す。
template <class T>
struct Holder:Base {
	T value{};
};

template <>
struct Holder<int>:Base {
	int value{};
};

// テンプレートクラスのサンプルを示す。
template <Numeric T>
class Accumulator final {
public:
	explicit Accumulator(T initial) : total_(initial) {}

	void add(T value)
	{
		total_ += value;
	}

	[[nodiscard]] T total() const noexcept
	{
		return total_;
	}

private:
	T total_{};
};

// MSVC 固有の __declspec 属性を付けた関数を示す。
__declspec(noinline) int normalize(int value) noexcept
{
	return value < 0 ? 0 : value;
}

// 自由関数と戻り値の型推論を示す。
[[nodiscard]] auto make_message(Status status) -> std::wstring
{
	switch (status) {
	case Status::ready:
		return L"ready";
	case Status::running:
		return L"running";
	case Status::finished:
		return L"finished";
	}
	return LR"(unknown)";   // RAW文字列のサンプル
}
