Static Sub CountCalls()
    Static callCount As Long

    callCount = callCount + 1
    Debug.Print callCount
End Sub

Public Function TryFindValue( _
    ByVal values As Variant, _
    ByVal expected As String, _
    ByRef foundIndex As Long _
) As Boolean
    Dim index As Long

    TryFindValue = False
    foundIndex = -1

    For index = LBound(values) To UBound(values)
        If StrComp(CStr(values(index)), expected, vbTextCompare) = 0 Then
            foundIndex = index
            TryFindValue = True
            Exit Function
        End If
    Next index
End Function

Private Function BuildMessage(ByVal name As String, ParamArray values() As Variant) As String
    Dim result As String
    Dim value As Variant

    result = name
    For Each value In values
        result = result & ": " & CStr(value)
    Next value

    BuildMessage = result
End Function
